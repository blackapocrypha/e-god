# 1 后台部署

服务器需要最低jdk17版本的Java环境，直接本机启动



start.sh

```sh
#! /bin/bash  
analysisor=deeeelete
program_name=EightGod
JAVA_OPTS="-Duser.timezone=GMT+8 -server -Xms256m -Xmx512m -Dlog4j2.formatMsgNoLookups=true"  
JAVA_OPTS="$JAVA_OPTS $@"  

nohup $JAVA_HOME/bin/java $JAVA_OPTS -jar  /usr/local/soft/EightGod.jar > ./eightGod-admin.log 2>&1 &
pid=$!  
exit_code=$?  
echo $pid > eightGod-admin-pid  
  
echo "pid: $pid"  
  
if [ $exit_code -eq 0 ];  
then
    echo "execute $program_name successfully!"
    echo "start real-time reading of $program_name logs"
    echo "ctrl + c exit"    
    tail -f eightGod-admin.log
else  
    echo "execute $program_name failed!"      
fi
```



stop.sh

```sh
#! /bin/bash  
author=deeeelete
program_name=EightGod
pid=`jps | grep EightGod | awk '{print $1}'`  
echo "start to kill pid:$pid"   
kill $pid  
if [ $? -eq 0 ];  
then  
    echo "shutdown $program_name successfully!"
else  
    echo "shutdown $program_name failed!"  
fi
```
